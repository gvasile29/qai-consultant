- [ ] **Step 2: Write the smoke-test script**

```python
import asyncio
import json
import sys
import time
from pathlib import Path

from mcp import ClientSession
from mcp.client.stdio import StdioServerParameters, stdio_client

REPO_ROOT = Path(r"C:\Personal\qai-consultant")
REVIEW_FIXTURE = REPO_ROOT / "evals" / "fixtures" / "review" / "strong_test_plan.md"
RESULTS_FIXTURES = [
    REPO_ROOT / "evals" / "fixtures" / "results" / f"run{i}.xml" for i in (1, 2, 3)
]

EXPECTED_TOOLS = {
    "retrieve_qa_knowledge", "list_kb_sources", "estimate_qa_effort",
    "review_qa_document", "analyze_test_results",
}

VALID_ESTIMATE_ARGS = {
    "project_name": "Sample Fintech API",
    "project_description": "Payment processing API for a fintech startup",
    "project_type": "api",
    "tech_stack": "Python, FastAPI, PostgreSQL",
    "team_qa_size": "3",
    "team_dev_size": "8",
    "timeline": "6 months",
    "methodology": "agile",
    "known_risks": "third-party payment gateway integration",
    "existing_automation": "some unit tests",
    "compliance_requirements": "PCI-DSS",
}


async def smoke_test():
    server_params = StdioServerParameters(command="qai-consultant-mcp", args=[])
    async with stdio_client(server_params) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()

            tools = await session.list_tools()
            names = {t.name for t in tools.tools}
            assert names == EXPECTED_TOOLS, f"unexpected tool set: {names}"
            print("list_tools OK:", sorted(names))

            result = await session.call_tool("list_kb_sources", {})
            assert not result.isError, result.content
            print("list_kb_sources OK")

            start = time.monotonic()
            result = await session.call_tool(
                "retrieve_qa_knowledge", {"query": "risk-based testing", "k": 3}
            )
            elapsed = time.monotonic() - start
            assert not result.isError, result.content
            print(f"retrieve_qa_knowledge OK (first embedding-model call: {elapsed:.1f}s)")

            review_doc = REVIEW_FIXTURE.read_text(encoding="utf-8")
            result = await session.call_tool(
                "review_qa_document", {"document_text": review_doc, "doc_type": "auto"}
            )
            assert not result.isError, result.content
            print("review_qa_document OK")

            multi_run = json.dumps([
                {"run_id": f"run{i}", "xml": p.read_text(encoding="utf-8")}
                for i, p in enumerate(RESULTS_FIXTURES, start=1)
            ])
            result = await session.call_tool("analyze_test_results", {"junit_xml": multi_run})
            assert not result.isError, result.content
            print("analyze_test_results (multi-run JSON-array string) OK")

            result = await session.call_tool("estimate_qa_effort", VALID_ESTIMATE_ARGS)
            assert not result.isError, result.content
            print("estimate_qa_effort OK")

    print("SMOKE TEST PASSED")


if __name__ == "__main__":
    asyncio.run(asyncio.wait_for(smoke_test(), timeout=180))
```
